% ----- Return buttons land on a slide's last step ----- %
% A frame's \label points at its first overlay, so "Return" to a slide that
% builds up would replay the build. Each labelled frame (mit-filters.lua) writes
% its page to the .aux on every overlay; the last write is its final step, and
% \returnbutton links to that page's beamer target. Falls back to the label on
% the first run.
\makeatletter
\newcommand{\setretpage}[2]{\expandafter\gdef\csname mit@retpage@#1\endcsname{#2}}
\newcommand{\markframeend}[1]{%
  \immediate\write\@auxout{\string\setretpage{#1}{\the\c@page}}}
\newcommand{\returnbutton}[2]{%
  \@ifundefined{mit@retpage@#1}%
    {\hyperref[#1]{\beamergotobutton{#2}}}%
    {\hyperlink{Navigation\csname mit@retpage@#1\endcsname}{\beamergotobutton{#2}}}}
\makeatother

% Main frames show n/N (title frame is frame 1: hidden and excluded from N).
% After \appendix the whole footline template is swapped for the A-n version
% (same approach as redefining the footline after \appendix in a plain
% metropolis deck). Standout frames suppress the footline.
\setbeamercolor{footline}{bg=MITDarkRed,fg=white}
% Footer text: set `footer:` in the document YAML (mit-filters.lua turns it into
% a \renewcommand); this is the fallback when the document doesn't set one.
\providecommand{\mitfootertext}{Joe Loffredo (MIT)}
\makeatletter
\newcommand{\mit@footline}[1]{%
  \ifbool{metropolis@standout}{}{%
  \leavevmode%
  \hbox{%
  \begin{beamercolorbox}[wd=.5\paperwidth,ht=2.5ex,dp=1.5ex,left,leftskip=2ex]{footline}%
    \usebeamerfont{page number in head/foot}\textbf{\mitfootertext}%
  \end{beamercolorbox}%
  \begin{beamercolorbox}[wd=.5\paperwidth,ht=2.5ex,dp=1.5ex,right,rightskip=2ex]{footline}%
    \usebeamerfont{page number in head/foot}\textbf{#1}%
  \end{beamercolorbox}}%
  \vskip0pt%
  }%
}
\setbeamertemplate{footline}{%
  \ifnum\insertframenumber=1\relax\else
    \mit@footline{\the\numexpr\insertframenumber-1\relax}%
  \fi
}

% ----- Appendix: A-1, A-2, ... page numbers (appendixnumberbeamer) ----- %
% The mit-filters.lua filter sets \ifmitappendixpending from inside the frame
% preceding the `# Appendix {#appendix}` divider; the section hook below runs
% at top level (between frames), where \appendix is safe to execute. The
% section page renders first (before \part can clobber the section title).
\usepackage{appendixnumberbeamer}
\newif\ifmitappendixpending
\AtBeginPart{}% pandoc's default hook renders a part page; \appendix calls \part
\AtBeginSection{%
  % metropolis's standout cleanup runs too late when a section page follows a
  % standout frame, leaking the maroon background canvas onto it. Its group is
  % still open here (an \endgroup would also revert the new section title), so
  % just override the canvas color; the group ends after this frame anyway.
  \ifbool{metropolis@standout}{\setbeamercolor{background canvas}{bg=white}}{}%
  \ifbeamer@inframe
    \sectionpage
  \else
    \frame[plain,c,noframenumbering]{\sectionpage}%
  \fi
  \ifmitappendixpending
    \global\mitappendixpendingfalse
    \appendix
    \setbeamertemplate{footline}{\mit@footline{A-\insertframenumber}}%
  \fi
}
\makeatother

% Uniform spacing across the deck, two rules:
%  1. Within a section, one gap: item to item, heading or lead-in to its first
%     item, and item to item across a pause (`. . .` splits a list in two) all
%     come out the same, because lists add no space above or below themselves
%     (topsep = partopsep = 0) and \parskip equals \itemsep.
%  2. Between sections, one larger gap: \sectiongap before every red section
%     heading except a slide's first.
% Pandoc's \tightlist (itemsep = parskip = 0) is neutralised so it cannot make
% some lists tighter than others.
% Slightly tighter line pitch deck-wide, so dense slides need no local fixes.
\linespread{0.94}
\setlength{\parskip}{4pt}
\setlength{\partopsep}{0pt}
\renewcommand{\tightlist}{}
\newcommand{\sectiongap}{\vspace{1.1em}}
% beamer calls the `body begin` templates BEFORE \list, which then resets
% \itemsep/\topsep from \@listi..\@listiii, so the list parameters themselves
% are what has to change. parsep stays 0 so a wrapped bullet does not open a
% gap inside itself.
\makeatletter
\renewcommand{\@listi}{%
  \leftmargin\leftmargini
  \topsep 0pt \partopsep 0pt \parsep 0pt \itemsep 4pt}
\let\@listI\@listi
\renewcommand{\@listii}{%
  \leftmargin\leftmarginii
  \labelwidth\leftmarginii \advance\labelwidth-\labelsep
  \topsep 0pt \partopsep 0pt \parsep 0pt \itemsep 2pt}
\renewcommand{\@listiii}{%
  \leftmargin\leftmarginiii
  \labelwidth\leftmarginiii \advance\labelwidth-\labelsep
  \topsep 0pt \partopsep 0pt \parsep 0pt \itemsep 2pt}
\makeatother

% \denselists: kept so existing slides still compile; the deck default is now
% this spacing, so it changes nothing.
\newcommand{\denselists}{}
\makeatletter
% \packedlists: no extra gap at all, for the one or two frames that carry more
% list than the default fits. Keeps the deck's type size.
\newcommand{\packedlists}{%
  \def\@listi{\leftmargin\leftmargini \topsep 0pt \partopsep 0pt \parsep 0pt \itemsep 0pt}%
  \let\@listI\@listi
  \def\@listii{%
    \leftmargin\leftmarginii
    \labelwidth\leftmarginii \advance\labelwidth-\labelsep
    \topsep 0pt \parsep 0pt \itemsep 0pt}%
  \def\@listiii{%
    \leftmargin\leftmarginiii
    \labelwidth\leftmarginiii \advance\labelwidth-\labelsep
    \topsep 0pt \parsep 0pt \itemsep 0pt}%
  \setlength{\parskip}{2pt}}
\makeatother
